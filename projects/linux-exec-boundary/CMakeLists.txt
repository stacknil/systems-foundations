cmake_minimum_required(VERSION 3.20)
project(linux_exec_boundary VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Native Linux is required; do not substitute Windows process semantics")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BOUNDARY_SANITIZERS "Instrument the supervisor with ASan/UBSan" OFF)
add_library(boundary_core src/supervisor.cpp src/manifest.cpp)
target_include_directories(boundary_core PUBLIC include)
add_executable(boundary_fixture fixtures/fixture.cpp)
target_include_directories(boundary_fixture PRIVATE include)
add_executable(exec_boundary src/main.cpp)
target_link_libraries(exec_boundary PRIVATE boundary_core)
target_compile_definitions(exec_boundary PRIVATE APPROVED_FIXTURE_PATH="$<TARGET_FILE:boundary_fixture>")
add_dependencies(exec_boundary boundary_fixture)
add_executable(capability_probe tools/capability_probe.cpp)
foreach(target boundary_core boundary_fixture exec_boundary capability_probe)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
if(BOUNDARY_SANITIZERS)
  target_compile_options(boundary_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(exec_boundary PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(exec_boundary PRIVATE -fsanitize=address,undefined)
endif()
add_executable(lifecycle_test tests/lifecycle.cpp)
target_link_libraries(lifecycle_test PRIVATE boundary_core)
target_compile_definitions(lifecycle_test PRIVATE APPROVED_FIXTURE_PATH="$<TARGET_FILE:boundary_fixture>")
target_compile_options(lifecycle_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_dependencies(lifecycle_test boundary_fixture)
if(BOUNDARY_SANITIZERS)
  target_link_options(lifecycle_test PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
set(QUALIFY_EXTRA)
if(BOUNDARY_SANITIZERS)
  list(APPEND QUALIFY_EXTRA --sanitized)
endif()
add_test(NAME lifecycle COMMAND lifecycle_test)
set_tests_properties(lifecycle PROPERTIES TIMEOUT 8)
add_test(NAME qualification COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/qualify.py
  --supervisor $<TARGET_FILE:exec_boundary> --probe $<TARGET_FILE:capability_probe>
  --output ${CMAKE_CURRENT_BINARY_DIR}/evidence ${QUALIFY_EXTRA})
set_tests_properties(qualification PROPERTIES TIMEOUT 50)
